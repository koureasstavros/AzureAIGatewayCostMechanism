        }" />
        <!-- 1. Read body ONCE into a string variable -->
        <set-variable name="responseBody" value="@(context.Response.Body.As<string>(preserveContent: true))" />
        <set-variable name="isStreamingResponse" value="@{
            string raw = (string)context.Variables["responseBody"];
            if (string.IsNullOrWhiteSpace(raw)) 
            { return false; }
            string trimmed = raw.TrimStart();
            return trimmed.Length == 0 || trimmed[0] != '{';
        }" />
        <set-variable name="responseJson" value="@{
            string raw = (string)context.Variables["responseBody"];
            if (string.IsNullOrWhiteSpace(raw))
            { return new JObject(); }
 
            bool isStreaming = (bool)context.Variables["isStreamingResponse"];
 
            if (!isStreaming)
            {
                // --- Non-streaming path: body IS the response object directly ---
                try { return JObject.Parse(raw); }
                catch { return new JObject(); }
            }
 
            // --- Streaming path: scan SSE "data: " lines for response.completed ---
            var lines = raw.Split('\n');
 
            foreach (var rawLine in lines)
            {
                var line = rawLine.TrimEnd('\r').Trim();
                if (!line.StartsWith("data: "))
                { continue; }
 
                var payload = line.Substring(6).Trim();
                if (payload == "[DONE]" || string.IsNullOrWhiteSpace(payload))
                { continue; }
 
                try
                {
                    var evt = JObject.Parse(payload);
                    var eventType = (string)evt["type"];
 
                    if (eventType == "response.completed")
                    {
                        // The actual response object (with model, usage, etc.) is nested under "response"
                        var inner = evt["response"] as JObject;
                        if (inner != null)
                        { return inner; }
                    }
                }
                catch
                {
                    // Partial/malformed line (e.g. obfuscated delta payloads) — skip and keep scanning
                    continue;
                }
            }
 
            // No response.completed found — return empty object so downstream ?? defaults kick in
            return new JObject();
        }" />
        <!-- 2. Extract token usage from the parsed (and now flat) response object -->
        <set-variable name="actualInputTokens" value="@{
            var body = (JObject)context.Variables["responseJson"];
            var inputTokens = body?["usage"]?["input_tokens"]?.Value<int>() ?? 0;
            return inputTokens;
        }" />
        <set-variable name="actualCachedReadInputTokens" value="@{
            var body = (JObject)context.Variables["responseJson"];
            var inputDetails = body?["usage"]?["input_tokens_details"] as JObject;
            var cachedTokens = inputDetails?["cached_tokens"]?.Value<int>() ?? 0;
            return cachedTokens;
        }" />
        <set-variable name="actualCachedWriteInputTokens" value="@{
            var body = (JObject)context.Variables["responseJson"];
            var inputDetails = body?["usage"]?["input_tokens_details"] as JObject;
            var cachedTokens = inputDetails?["cache_write_tokens"]?.Value<int>() ?? 0;
            return cachedTokens;
        }" />
        <set-variable name="actualOutputTokens" value="@{
            var body = (JObject)context.Variables["responseJson"];
            var outputTokens = body?["usage"]?["output_tokens"]?.Value<int>() ?? 0;
            return outputTokens;
        }" />
        <!-- 3. Load existing counters from cache -->
        <cache-lookup-value key="@($"quota/{context.Subscription.Key}/cost/{context.Variables["yearMonth"]}")" variable-name="storedCost" />
        <!-- 4. Cost rates -->
        <set-variable name="inputCostPerM" value="@{

            if (model.IndexOf("GPT56TERRA-GS", StringComparison.OrdinalIgnoreCase) >= 0)

            if (model.IndexOf("GPT56SOL-GS", StringComparison.OrdinalIgnoreCase) >= 0)
            { return 5.0; }
            if (model.IndexOf("KIMIK25-GS", StringComparison.OrdinalIgnoreCase) >= 0)
            { return 0.5; }
            if (model.IndexOf("KIMIK26-GS", StringComparison.OrdinalIgnoreCase) >= 0)
            { return 0.95; }
            return 5.0; // Default fallback
        }" />
        <set-variable name="inputCachedReadCostPerM" value="@{
            var model = (string)context.Variables["modelId"];
            if (string.IsNullOrEmpty(model))
            { return 0.0; }            
            if (model.IndexOf("GPT54NANO-GS", StringComparison.OrdinalIgnoreCase) >= 0)
            { return 0.02; }
            if (model.IndexOf("GPT54MINI-GS", StringComparison.OrdinalIgnoreCase) >= 0)
            { return 0.075; }
            if (model.IndexOf("GPT54-GS", StringComparison.OrdinalIgnoreCase) >= 0)
            { return 0.25; }

            { return 0.5; }
            if (model.IndexOf("GPT56LUNA-GS", StringComparison.OrdinalIgnoreCase) >= 0)
            { return 0.1; }
            if (model.IndexOf("GPT56TERRA-GS", StringComparison.OrdinalIgnoreCase) >= 0)
            { return 0.25; }
            if (model.IndexOf("GPT55SOL-GS", StringComparison.OrdinalIgnoreCase) >= 0)
            { return 0.5; }
			if (model.IndexOf("KIMIK25-GS", StringComparison.OrdinalIgnoreCase) >= 0)
            { return 0.1; }
            if (model.IndexOf("KIMIK26-GS", StringComparison.OrdinalIgnoreCase) >= 0)
            { return 0.16; }
            return 1.0; // Default fallback            
        }" />
        <set-variable name="inputCachedWriteCostPerM" value="@{
            var model = (string)context.Variables["modelId"];
            if (string.IsNullOrEmpty(model))
            { return 0.0; }
            if (model.IndexOf("GPT54NANO-GS", StringComparison.OrdinalIgnoreCase) >= 0)
            { return 0.25; }
            if (model.IndexOf("GPT54MINI-GS", StringComparison.OrdinalIgnoreCase) >= 0)

            return 30.0; // Default fallback
        }" />
        <!-- 5. Calculate cost -->
        <set-variable name="inputCost" value="@{
            double rate = (double)context.Variables["inputCostPerM"];
			double tokens = (double)(int)context.Variables["actualInputTokens"];
			double cachedreadrate = (double)context.Variables["inputCachedReadCostPerM"];
			double cachedreadtokens = (double)(int)context.Variables["actualCachedReadInputTokens"];
			double cachedwriterate = (double)context.Variables["inputCachedWriteCostPerM"];
			double cachedwritetokens = (double)(int)context.Variables["actualCachedWriteInputTokens"];
            double cost = rate * (tokens - cachedreadtokens - cachedwritetokens) / 1000000.0;
            double cachedreadcost = cachedreadrate * cachedreadtokens / 1000000.0;
            double cachedwritecost = cachedwriterate * cachedwritetokens / 1000000.0;
			return cost + cachedreadcost + cachedwritecost;
        }" />
        <set-variable name="outputCost" value="@{
            double rate = (double)context.Variables["outputCostPerM"];
			double tokens = (double)(int)context.Variables["actualOutputTokens"];
            double cost = rate * tokens / 1000000.0;
			return cost;
        }" />
        <!-- 6. Calculate current cost -->
        <set-variable name="currentCost" value="@{
            double inputCost = (double)context.Variables["inputCost"];
            double outputCost = (double)context.Variables["outputCost"];
            return inputCost + outputCost;
        }" />
        <!-- 7. Increment counters -->
        <set-variable name="newCostTotal" value="@{
            double previous = 0;
            if (context.Variables.ContainsKey("storedCost")) {
                previous = (double)context.Variables["storedCost"];
            }
            double current = (double)context.Variables["currentCost"];
            return previous + current;
        }" />
        <!-- 8. Store updated counters back into cache (30 days) -->
        <cache-store-value key="@($"quota/{context.Subscription.Key}/cost/{context.Variables["yearMonth"]}")" value="@((double)context.Variables["newCostTotal"])" duration="2592000" />

        </emit-metric>

                double v = 0.0;
                if (context.Variables.ContainsKey("currentCost"))
                { v = Convert.ToDouble(context.Variables["currentCost"]); }
                return v;
            }">

            <dimension name="Status Code" value="@(context.Response.StatusCode.ToString())" />
            <dimension name="Model ID" value="@(context.Variables.GetValueOrDefault("modelId",""))" />
        </emit-metric>
    </outbound>
    <on-error>
        <base />
    </on-error>
